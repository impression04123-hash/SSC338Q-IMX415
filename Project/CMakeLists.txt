cmake_minimum_required(VERSION 3.16)
project(ssc338q_fpv VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
option(FPV_ENABLE_SIGMASTAR "Build the SigmaStar MI hardware capture backend" OFF)
option(FPV_ENABLE_HISILICON "Build the Hi3536DV100 MPP display backend" OFF)
option(FPV_SANITIZE "Enable address and undefined behavior sanitizers on host" OFF)
include(CTest)
add_library(fpv_core STATIC src/common/rtp.c src/common/config.c src/common/io.c)
target_include_directories(fpv_core PUBLIC include)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(fpv_core PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
  if(FPV_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
    target_compile_options(fpv_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
endif()
include(cmake/platforms.cmake)
fpv_add_platforms()
add_executable(fpvctl src/ctl/fpvctl.c)
target_link_libraries(fpvctl PRIVATE fpv_core)
add_executable(fpv-air src/air/main.c)
target_link_libraries(fpv-air PRIVATE fpv_core)
add_executable(fpv-ground src/ground/main.c)
target_link_libraries(fpv-ground PRIVATE fpv_core)
foreach(app IN ITEMS fpv-air fpv-ground fpvctl)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${app} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
  endif()
endforeach()
if(FPV_ENABLE_SIGMASTAR)
  target_compile_definitions(fpv-air PRIVATE FPV_HAVE_SIGMASTAR=1)
  target_link_libraries(fpv-air PRIVATE fpv_sigmastar)
endif()
if(FPV_ENABLE_HISILICON)
  target_compile_definitions(fpv-ground PRIVATE FPV_HAVE_HISILICON=1)
  target_link_libraries(fpv-ground PRIVATE fpv_hisilicon)
endif()
install(TARGETS fpv-air fpv-ground fpvctl RUNTIME DESTINATION bin)
install(DIRECTORY configs/ DESTINATION /etc/fpv FILES_MATCHING PATTERN "*.ini")
install(FILES LICENSE README.md THIRD_PARTY.md DESTINATION share/fpv)
install(DIRECTORY docs/ DESTINATION share/fpv/docs FILES_MATCHING PATTERN "*.md")
install(PROGRAMS scripts/init.d/S95fpv DESTINATION share/fpv)
if(BUILD_TESTING)
  add_executable(test_rtp tests/test_rtp.c)
  target_link_libraries(test_rtp PRIVATE fpv_core)
  add_test(NAME rtp_protocol COMMAND test_rtp)
  add_executable(test_rtp_extra tests/test_rtp_extra.c)
  target_link_libraries(test_rtp_extra PRIVATE fpv_core)
  add_test(NAME rtp_edge_cases COMMAND test_rtp_extra)
  add_test(NAME rtp_invalid_packetizer COMMAND test_rtp_extra packetizer-overflow)
  add_executable(test_rtp_fuzz tests/test_rtp_fuzz.c)
  target_link_libraries(test_rtp_fuzz PRIVATE fpv_core)
  add_test(NAME rtp_stress COMMAND test_rtp_fuzz)
  add_executable(test_config tests/test_config.c)
  target_link_libraries(test_config PRIVATE fpv_core)
  add_test(NAME config_validation COMMAND test_config)
  add_executable(test_config_extra tests/test_config_extra.c)
  target_link_libraries(test_config_extra PRIVATE fpv_core)
  add_test(NAME config_edge_cases COMMAND test_config_extra ${CMAKE_CURRENT_SOURCE_DIR})
  add_executable(test_capture_annexb tests/hardware/test_capture_annexb.c)
  target_include_directories(test_capture_annexb PRIVATE src/platform include)
  add_test(NAME hardware_annexb COMMAND test_capture_annexb)
  add_executable(test_io tests/test_io.c)
  target_link_libraries(test_io PRIVATE fpv_core)
  add_test(NAME media_io COMMAND test_io)
  if(NOT CMAKE_CROSSCOMPILING)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
      add_test(NAME runtime_supervisor COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime/test_runtime.py $<TARGET_FILE:fpvctl>)
      add_test(NAME video_pipeline COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_pipeline.py $<TARGET_FILE_DIR:fpv-air>)
      add_test(NAME air_metadata COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_air_metadata.py $<TARGET_FILE:fpv-air>)
      add_test(NAME restart_session COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_restart_session.py $<TARGET_FILE_DIR:fpv-air>)
      set_tests_properties(video_pipeline PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
    endif()
  endif()
endif()
